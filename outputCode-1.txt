const fs = require('fs');
const PDFDocument = require('pdfkit');

function calculateChargingCost(initialKwh, finalKwh, costPerKwh) {
    const kwhUsed = finalKwh - initialKwh;
    const totalCost = kwhUsed * costPerKwh;
    return totalCost;
}

function createPDF(cost, filename) {
    const doc = new PDFDocument();
    doc.pipe(fs.createWriteStream(filename));
    
    doc.fontSize(25).text('Charging Cost Report', { align: 'center' });
    doc.moveDown();
    doc.fontSize(16).text(`Total Charging Cost: ${cost.toFixed(2)} ETB`, { align: 'center' });
    
    doc.end();
}

const initialKwh = 100; // Example initial reading
const finalKwh = 150; // Example final reading
const costPerKwh = 5; // Example cost per kWh in ETB

const totalCost = calculateChargingCost(initialKwh, finalKwh, costPerKwh);
createPDF(totalCost, 'ADDIS_GREEN_EV_SOLUTION_S.C..pdf');
